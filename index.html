Supabase SQL セットアップ・解説

Supabaseのダッシュボードメニュー SQL Editor を開き、以下のクエリを貼り付けて Run を実行してください。プロフィール同期とリアルタイムメッセージ機能が有効化されます。 

-- ==========================================
-- 1. ユーザープロフィール用テーブル (profiles)
-- ==========================================
-- ユーザーID、名前、母国語、学習言語、自己紹介、アイコン画像URLを保持します
CREATE TABLE IF NOT EXISTS profiles (
    id TEXT PRIMARY KEY,
    name TEXT NOT NULL,
    native_lang TEXT NOT NULL,
    target_lang TEXT NOT NULL,
    bio TEXT,
    avatar_url TEXT,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now())
);

-- ==========================================
-- 2. リアルタイムメッセージ用テーブル (messages)
-- ==========================================
-- 送信者/受信者IDとチャット本文、送信日時を記録します
CREATE TABLE IF NOT EXISTS messages (
    id BIGSERIAL PRIMARY KEY,
    sender_id TEXT NOT NULL,
    receiver_id TEXT NOT NULL,
    content TEXT NOT NULL,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now())
);

-- ==========================================
-- 3. Row Level Security (RLS) アクセス制御設定
-- ==========================================
-- すべてのユーザーがデータの読み書き（交流）を行えるよう全許可ポリシーを設定します
ALTER TABLE profiles ENABLE ROW LEVEL SECURITY;
ALTER TABLE messages ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Allow all public read/write for profiles" ON profiles FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Allow all public read/write for messages" ON messages FOR ALL USING (true) WITH CHECK (true);

-- ==========================================
-- 4. Realtime（リアルタイム同期）の有効化
-- ==========================================
-- 他のデバイスやブラウザで即座にメッセージが更新されるよう設定します
ALTER PUBLICATION supabase_realtime ADD TABLE profiles;
ALTER PUBLICATION supabase_realtime ADD TABLE messages;
※ボタンを押すとSQL全文がコピーされます
